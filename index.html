🎨 Google Apps Script Auto-Colorizer

Since Google Calendar forces imported ICS events to default blue, run this 10-second script in your Google Account to automatically recolor all events across all 6 weeks:

  1. Go to script.google.com and click New Project.
  2. Paste the script below, replace the placeholder text, and click Run.
function colorizeAcademicCalendar() {
  var cal = CalendarApp.getCalendarById("your_email@gmail.com"); 
  var startDate = new Date("2026-10-12");
  var endDate = new Date("2026-11-23");
  var events = cal.getEvents(startDate, endDate);
  
  events.forEach(function(e) {
    var title = e.getTitle();
    if (title.indexOf("[Marketing]") !== -1) {
      e.setColor(CalendarApp.EventColor.RED); // Tomato (11)
    } else if (title.indexOf("[Data Analytics]") !== -1) {
      e.setColor(CalendarApp.EventColor.BLUE); // Blueberry (9)
    } else if (title.indexOf("[Macroeconomics]") !== -1) {
      e.setColor(CalendarApp.EventColor.GREEN); // Basil (10)
    } else if (title.indexOf("[Group Project]") !== -1) {
      e.setColor(CalendarApp.EventColor.ORANGE); // Tangerine (6)
    } else if (title.indexOf("[Buffer]") !== -1) {
      e.setColor(CalendarApp.EventColor.MAUVE); // Grape (3)
    }
  });
  Logger.log("Successfully recolored " + events.length + " events!");
}
